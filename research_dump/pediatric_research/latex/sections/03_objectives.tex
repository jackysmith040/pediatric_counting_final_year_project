\section{Research Objectives}

\begin{frame}{Primary Objective}
    To develop an occlusion-resistant computer vision framework for accurate pediatric patient counting in high-density medical waiting rooms, specifically addressing the ``Invisible Child'' scenario (infants carried on backs).
\end{frame}

\begin{frame}{Specific Objectives}
    \begin{itemize}
        \item \textbf{Develop an Allometric Scaling Model:} Establish a robust Head-to-Body ratio ($R$) classifier.
        \item \textbf{Implement Euclidean Centroid Tracking:} Utilize DeepSORT for ID persistence.
        \item \textbf{Optimize with Instance Segmentation:} Integrate YOLOv8-seg for pixel-level mask precision.
        \item \textbf{Validate in Local Context:} Fine-tune the model reflecting Ghanaian clinical settings.
    \end{itemize}
\end{frame}
