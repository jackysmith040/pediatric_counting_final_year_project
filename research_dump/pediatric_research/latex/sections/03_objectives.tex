\section{Objectives}

\begin{frame}{Objectives of the Study}
    \textbf{General Objective:} Develop a privacy-preserving, automated mathematical framework to accurately count pediatric visitors, including occluded infants.
    
    \vspace{0.4cm}
    \textbf{Specific Objectives:}
    \begin{enumerate}
        \item \textbf{Automated Detection:} Formulate a spatial model to mathematically separate carried children from adult caregivers.
        \item \textbf{Mathematical Anthropometry:} Apply physical constraints via Head-to-Body Ratios to prevent misclassification.
        \item \textbf{Geometric Tracking:} Implement centroid tracking utilizing Euclidean distance to ensure non-repeating counts.
        \item \textbf{Edge Architecture:} Design a localized computing system that outputs strictly numerical matrices.
    \end{enumerate}
\end{frame}
