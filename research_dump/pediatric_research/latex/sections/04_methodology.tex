\section{Methodology}

\begin{frame}{Methodology I: Implementation Stack}
    We utilize a modular pipeline optimized for clinical edge deployment:
    \begin{itemize}
        \item \textbf{Detection:} \textbf{YOLOv11-seg} for pixel-level precision masks.
        \item \textbf{Validation:} \textbf{Ada-Haar Booster} to verify pediatric head signatures.
        \item \textbf{Tracking:} \textbf{DeepSORT} for ID persistence across occlusions.
    \end{itemize}
\end{frame}

\begin{frame}{Methodology II: Allometric Logic}
    To solve the ``Invisible Child'' dyad, we evaluate spatial invariants:
    \begin{itemize}
        \item \textbf{Ratio Calculation:} $R = H_{total} / H_{head} \approx 4.0 - 6.0$.
        \item \textbf{Vertical Displacement:} $y_{h2} < y_{h1}$ (Child on back).
        \item \textbf{Area Proportion:} $A_{h2} < A_{h1}$ (Pediatric scale check).
    \end{itemize}

\begin{frame}{Methodology III: Kalman Filtering}
    To maintain patient IDs during high-density overlaps, we utilize linear state prediction:
    \begin{equation}
        \hat{x}_k^- = F\hat{x}_{k-1} + Bu_k
    \end{equation}
    This ensures that a mother and child detected as a single cluster are mathematically tracked as independent biological entities.
\end{frame}
