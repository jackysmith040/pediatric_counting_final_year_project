\section{Methodology}

\begin{frame}{Proposed Pipeline}
    The proposed system follows a 4-stage pipeline:
    \begin{center}
        \textbf{Detection $\to$ Segmentation $\to$ Tracking $\to$ Classification}
    \end{center}
\end{frame}

\begin{frame}{Object Detection \& Segmentation}
    \begin{itemize}
        \item \textbf{Model:} YOLOv8-seg.
        \item \textbf{Refinement (Booster):} AdaBoost-Haar Cascade classifier.
        \item \textbf{Function:} YOLOv8-seg detects human segments, while the \textbf{Ada-Haar Booster} validates pediatric head signatures within overlapping regions.
    \end{itemize}
\end{frame}

\begin{frame}{Centroid Tracking (DeepSORT)}
    \begin{itemize}
        \item \textbf{Mechanism:} Calculates Euclidean distance between centroids.
        \item \textbf{Kalman Filter:} Predicts future positions to maintain ID persistence during occlusions.
    \end{itemize}
\end{frame}

\begin{frame}{Allometric Classification (The R Ratio)}
    $$ R = \frac{H_{total}}{H_{head}} $$
    \begin{block}{Classification Thresholds}
        \begin{itemize}
            \item $R \in [4.0, 6.0] \to$ \textbf{Infant}
            \item $R \in [7.0, 8.0] \to$ \textbf{Adult}
        \end{itemize}
    \end{block}
\end{frame}
