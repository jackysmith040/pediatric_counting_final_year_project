\section{The Search for a Reliable Census}

\begin{frame}{Beyond the Clicker}
    \begin{itemize}
        \item \textbf{The Question:} Can we automate the pediatric census in high-stakes environments like KNUST OPD?
        \item \textbf{The Baseline:} Manual ``clickers'' are slow, error-prone, and operationally expensive.
    \end{itemize}
\end{frame}

\begin{frame}{The False Start: Why Standard AI Fails}
    We assumed YOLO would work. We discovered a systemic blindness:
    \begin{itemize}
        \item \textbf{Scale Blindness:} Infants ignored as ``background noise.''
        \item \textbf{Geometric Occlusion:} The ``Invisible Child'' (infants on backs) absorbed into adult bounding boxes.
    \end{itemize}
    \begin{alertblock}{The Realization}
        Current Computer Vision is structurally biased against African clinical norms.
    \end{alertblock}
\end{frame}
