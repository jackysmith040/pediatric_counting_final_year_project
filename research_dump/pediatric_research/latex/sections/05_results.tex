\section{Analysis and Expected Results}

\begin{frame}{Precision Calibration}
    The system's precision is evaluated using the Mean Absolute Percentage Error (MAPE):
    \begin{equation}
        MAPE = \frac{1}{n} \sum_{i=1}^{n} \left| \frac{Actual_i - Predicted_i}{Actual_i} \right| \times 100
    \end{equation}
\end{frame}

\begin{frame}{Expected Operational Outcomes}
    \begin{itemize}
        \item \textbf{High-Precision Classification:} Capture over 90\% of pediatric visitors, including those occluded by caregivers.
        \item \textbf{Operational Trigger:} If pediatric density exceeds a threshold (e.g., 30\% occupancy), staff are alerted to open triage rooms.
        \item \textbf{Privacy Guarantee:} Processing occurs entirely in volatile memory; strictly numerical output.
    \end{itemize}
\end{frame}
