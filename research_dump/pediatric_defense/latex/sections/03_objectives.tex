\begin{frame}{Objectives: Restoring Visibility}
    \textbf{Primary Goal:} Develop an automated, privacy-preserving mathematical framework to accurately count pediatric visitors in high-density environments.
    
    \vspace{0.5cm}
    \textbf{Specific Objectives:}
    \begin{enumerate}
        \item \textbf{Mathematical Anthropometry:} Use biological scaling (Head-to-Body ratio) to break the geometric limits of standard AI bounding boxes.
        \item \textbf{Spatial Inference:} Develop a spatial logic engine to mathematically separate carried children from their adult caregivers without relying on facial recognition.
        \item \textbf{Neuro-Symbolic Tracking:} Deploy Kalman Filtering kinematics to ensure unique, non-repeating patient IDs even when subjects are temporarily occluded.
        \item \textbf{Edge Computing:} Process all video data locally (in RAM) to output strictly numerical density matrices, guaranteeing absolute medical privacy.
    \end{enumerate}
\end{frame}
