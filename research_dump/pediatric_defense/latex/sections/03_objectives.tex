\begin{frame}{Objectives: Restoring Visibility}
    \textbf{Primary Goal:} Develop an automated, privacy-preserving mathematical framework to accurately count pediatric visitors.
    
    \vspace{0.5cm}
    \textbf{Specific Objectives:}
    \begin{enumerate}
        \item \textbf{Mathematical Anthropometry:} Use biological scaling (Head-to-Body ratio) to break the geometric limits of standard AI.
        \item \textbf{Spatial Inference:} Mathematically separate carried children from their adult caregivers.
        \item \textbf{Neuro-Symbolic Tracking:} Deploy Kalman Filtering to ensure unique, non-repeating patient IDs.
        \item \textbf{Edge Computing:} Process data locally (in RAM) to guarantee strict medical privacy.
    \end{enumerate}
\end{frame}
