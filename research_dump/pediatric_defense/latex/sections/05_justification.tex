\begin{frame}{Justification: Why This Matters}
    \textbf{Operational Efficiency:}
    \begin{itemize}
        \item Manual clickers and tallies are prone to fatigue and error.
        \item Real-time automated data allows administrators to dynamically shift staff during peak hours.
    \end{itemize}
    
    \vspace{0.5cm}
    \textbf{Ethical Medical AI:}
    \begin{itemize}
        \item \textbf{Privacy First:} By processing data on the "Edge" (volatile memory), no video is saved. We only output numerical matrices.
        \item \textbf{Contextual Design:} We aren't forcing a Western dataset onto a Ghanaian problem. We are using biological math to solve a cultural geometric challenge.
    \end{itemize}
\end{frame}
